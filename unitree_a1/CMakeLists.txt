cmake_minimum_required(VERSION 3.5)
project(talon_a1_driver CXX)

set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(TALON_A1_WITH_SDK "Link unitree_legged_sdk (real robot). OFF builds the loopback desk plant." ON)
set(UNITREE_SDK_DIR "" CACHE PATH "unitree_legged_sdk v3.3.x checkout (contains include/ and lib/)")

find_package(Threads REQUIRED)
find_package(pybind11 CONFIG REQUIRED)  # pip install pybind11, then -Dpybind11_DIR=$(python3 -m pybind11 --cmakedir)

# NO_EXTRAS: no LTO. GCC's LTO wrapper breaks on output paths containing a quote ("Master's Degree").
pybind11_add_module(_a1_driver NO_EXTRAS driver/a1_driver.cpp)
target_include_directories(_a1_driver PRIVATE driver)
target_link_libraries(_a1_driver PRIVATE Threads::Threads)
# The module lands inside the Python package so `import talon_a1` finds it without installing.
set_target_properties(_a1_driver PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/talon_a1)

if(TALON_A1_WITH_SDK)
  if(NOT UNITREE_SDK_DIR)
    message(FATAL_ERROR "Set -DUNITREE_SDK_DIR=<unitree_legged_sdk checkout>, or -DTALON_A1_WITH_SDK=OFF for the desk plant")
  endif()
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
    set(_sdk_arch amd64)
  elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
    set(_sdk_arch arm64)
  else()
    set(_sdk_arch arm32)
  endif()
  find_library(UNITREE_SDK_LIB NAMES unitree_legged_sdk_${_sdk_arch} PATHS ${UNITREE_SDK_DIR}/lib NO_DEFAULT_PATH)
  if(NOT UNITREE_SDK_LIB)
    message(FATAL_ERROR "libunitree_legged_sdk_${_sdk_arch}.so not found in ${UNITREE_SDK_DIR}/lib")
  endif()
  # The SDK library has undefined lcm_* symbols even though the driver calls no LCM API.
  find_library(LCM_LIB NAMES lcm)
  if(NOT LCM_LIB)
    message(FATAL_ERROR "liblcm not found; the SDK library needs it")
  endif()
  message(STATUS "talon_a1: linking ${UNITREE_SDK_LIB} ${LCM_LIB}")
  target_include_directories(_a1_driver PRIVATE ${UNITREE_SDK_DIR}/include)
  target_link_libraries(_a1_driver PRIVATE ${UNITREE_SDK_LIB} ${LCM_LIB})
  target_compile_definitions(_a1_driver PRIVATE TALON_A1_WITH_SDK=1)
  set_target_properties(_a1_driver PROPERTIES BUILD_RPATH ${UNITREE_SDK_DIR}/lib)
else()
  message(STATUS "talon_a1: loopback desk plant (no robot)")
endif()

enable_testing()
add_executable(test_safety_core driver/test_safety_core.cpp)
add_test(NAME safety_core COMMAND test_safety_core)
